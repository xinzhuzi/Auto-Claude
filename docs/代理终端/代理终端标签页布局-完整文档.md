# 代理终端 - 浏览器标签页式布局改造计划

## 📋 项目概述

**目标**: 将代理终端的布局从"网格式全部展示"改造为"浏览器标签页式"，每个终端占满当前界面布局区域，通过标签页切换。

**当前状态分析**:
- 文件位置: `apps/frontend/src/renderer/components/TerminalGrid.tsx`
- 当前布局: 使用 `react-resizable-panels` 实现网格布局，网格最大支持 3x4 布局（显示12个终端）
- 实际限制: `terminal-store.ts` 中 `maxTerminals: Infinity`，理论上无限制
- 布局逻辑: 根据终端数量动态计算行列数 (1个=1x1, 2个=1x2, 4个=2x2, 最大3x4)
- 拖拽功能: 使用 `@dnd-kit` 实现终端重排序和文件拖放

**改造后的优势**:
- ✅ 突破网格布局的 12 个终端显示限制
- ✅ 标签页可以无限滚动，支持更多终端
- ✅ 每个终端都能获得完整的显示区域

## 🎯 核心需求

### 功能需求
1. **横向滑动标签页控件**:
   - 替换工具栏区域的 "X / 12 terminals" 提示
   - 横向滑动显示所有打开的终端标签
   - 每个标签显示终端名称（支持自定义名称或默认名称）
   - 标签过多时支持横向滚动
2. **单终端占满布局**: 当前激活的终端占满代理终端界面的内容区域（不是全屏，是当前界面布局）
3. **标签页切换**: 点击标签页切换终端
4. **标签页管理**:
   - 关闭按钮 (X)
   - 拖拽重排序
   - 新建终端按钮 (+)
   - 终端重命名功能
5. **保留现有功能**:
   - 文件拖放到终端
   - Claude 模式指示
   - Worktree 关联
   - 终端状态显示

### UI/UX 需求
1. **视觉风格**: 类似 VS Code / Chrome 的标签页
2. **响应式**: 标签页过多时支持滚动
3. **快捷键**:
   - `Ctrl/Cmd + T`: 新建终端
   - `Ctrl/Cmd + W`: 关闭当前终端
   - `Ctrl/Cmd + Tab`: 切换到下一个终端
   - `Ctrl/Cmd + 数字`: 切换到指定终端

#### TerminalTab.tsx (新组件)

```typescript
interface TerminalTabProps {
  terminal: Terminal;
  isActive: boolean;
  onSelect: () => void;
  onClose: () => void;
  onRename: (newName: string) => void;
  dragHandleProps?: any;
}

export function TerminalTab({
  terminal,
  isActive,
  onSelect,
  onClose,
  onRename,
  dragHandleProps,
}: TerminalTabProps) {
  const [isEditing, setIsEditing] = useState(false);
  const [editName, setEditName] = useState(terminal.title);

  const handleDoubleClick = () => {
    setIsEditing(true);
  };

  const handleBlur = () => {
    setIsEditing(false);
    if (editName.trim() && editName !== terminal.title) {
      onRename(editName.trim());
    } else {
      setEditName(terminal.title);
    }
  };

  const handleKeyDown = (e: React.KeyboardEvent) => {
    if (e.key === 'Enter') {
      handleBlur();
    } else if (e.key === 'Escape') {
      setIsEditing(false);
      setEditName(terminal.title);
    }
  };

  return (
    <div
      className={cn(
        "flex items-center gap-2 px-3 py-2 border-r border-border cursor-pointer",
        "hover:bg-accent/50 transition-colors min-w-[120px] max-w-[200px]",
        isActive && "bg-background border-b-2 border-b-primary"
      )}
      onClick={onSelect}
      {...dragHandleProps}
    >
      {/* 状态指示器 */}
      <div className="flex items-center gap-1 shrink-0">
        {terminal.isClaudeMode && (
          <Sparkles className="h-3 w-3 text-primary" />
        )}
        {terminal.worktreeConfig && (
          <GitBranch className="h-3 w-3 text-info" />
        )}
      </div>

      {/* 终端名称 - 可编辑 */}
      {isEditing ? (
        <input
          type="text"
          value={editName}
          onChange={(e) => setEditName(e.target.value)}
          onBlur={handleBlur}
          onKeyDown={handleKeyDown}
          className="flex-1 text-sm bg-transparent border-none outline-none focus:ring-1 focus:ring-primary rounded px-1"
          autoFocus
          onClick={(e) => e.stopPropagation()}
        />
      ) : (
        <span
          className="flex-1 text-sm truncate"
          onDoubleClick={handleDoubleClick}
          title={terminal.title}
        >
          {terminal.title}
        </span>
      )}

      {/* 关闭按钮 */}
      <Button
        variant="ghost"
        size="icon"
        classN"h-4 w-4 p-0 shrink-0 hover:bg-destructive/20 opacity-0 group-hover:opacity-100"
        onClick={(e) => {
          e.stopPropagation();
          onClose();
        }}
      >
        <X className="h-3 w-3" />
      </Button>
    </div>
  );
}
```

### 1. 拖拽实现

使用 `@dnd-kit` 的 `useSortable` hook:

```typescript
// SortableTerminalTab.tsx
export function SortableTerminalTab({ terminal, ...props }: Props) {
  const {
    attributes,
    listeners,
    setNodeRef,
    transform,
    transition,
    isDragging,
  } = useSortable({
    id: terminal.id,
    data: {
      type: 'terminal-tab',
      terminalId: terminal.id,
    },
  });

  const style = {
    transform: CSS.Transform.toString(transform),
    transition,
  };

  return (
    <div ref={setNodeRef} style={style} {...attributes}>
      <TerminalTab
        terminal={terminal}
        dragHandleProps={listeners}
        isDragging={isDragging}
        {...props}
      />
    </div>
  );
}
```

### 2. 快捷键实现

```typescript
// TerminalGrid.tsx
useEffect(() => {
  if (!isActive) return;

  const handleKeyDown = (e: KeyboardEvent) => {
   od = e.ctrlKey || e.metaKey;
n    // Ctrl/Cmd + Tab: 下一个终端
    if (isMod && e.key === 'Tab' && !e.shiftKey) {
      e.preventDefault();
      const currentIndex = terminals.findIndex(t => t.id === activeTerminalId);
      const nextIndex = (currentIndex + 1) % terminals.length;
      setActiveTerminal(terminals[nextIndex].id);
    }

    // Ctrl/Cmd + Shift + Tab: 上一个终端
    if (isMod && e.key === 'Tab' && e.shiftKey) {
      e.preventDefault();
      const currentIndex = terminals.findIndex(t => t.id === activeTerminalId);
      const prevIndex = (currentIndex - 1 + terminals.length) % terminals.length;
      setActiveTerminal(terminals[prevIndex].id);
    }

    // Ctrl/Cmd + 数字: 切换到指定终端
    if (isMod && /^[1-9]$/.test(e.key)) {
      e.preventDefault();
      const index = parseInt(e.key) - 1;
      if (index < terminals.length) {
        setActiveTerminal(terminals[index].id);
      }
    }
  };

  window.addEventListener('keydown', handleKeyDown);
  return () => window.removeEventListener('keydown', handleKeyDown);
}, [isActive, terminals, activeTerminalId, setActiveTerminal]);
```
