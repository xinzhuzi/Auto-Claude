> 可以直接复制使用的脚本

## 1. 完整合并脚本

保存为 `merge-upstream.sh`，放在项目根目录执行：

```bash
#!/bin/bash
# 合并上游脚本
# 使用: ./merge-upstream.sh

set -e  # 遇到错误停止

echo "========================================"
echo "Auto-Claude 上游合并脚本"
echo "========================================"

# 颜色定义
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m' # No Color

# 项目目录
PROJECT_DIR="/Users/zhengbingjin/Project/Github/Auto-Claude"
UPSTREAM_DIR="/Users/zhengbingjin/Project/temp/Auto-Claude"
BACKUP_BRANCH="backup/pre-merge-$(date +%Y%m%d-%H%M%S)"

cd "$PROJECT_DIR"

# 1. 检查工作目录
echo -e "${YELLOW}[1/7] 检查工作目录...${NC}"
if ! git diff-index --quiet HEAD --; then
    echo -e "${RED}工作目录有未提交的修改！${NC}"
    echo "请先提交或暂存修改"
    git status
    exit 1
fi
echo -e "${GREEN}✓ 工作目录干净${NC}"

# 2. 创建备份分支
echo -e "${YELLOW}[2/7] 创建备份分支: $BACKUP_BRANCH${NC}"
git branch "$BACKUP_BRANCH"
echo -e "${GREEN}✓ 备份创建完成${NC}"

# 3. 添加上游
echo -e "${YELLOW}[3/7] 添加上游源...${NC}"
git remote remove local-upstream 2>/dev/null || true
git remote add local-upstream "$UPSTREAM_DIR"
git fetch local-upstream
echo -e "${GREEN}✓ 上游添加完成${NC}"

# 4. 预览差异
echo -e "${YELLOW}[4/7] 预览差异...${NC}"
COMMIT_COUNT=$(git log develop..local-upstream/develop --oneline | wc -l)
echo "待合并提交数: $COMMIT_COUNT"
echo ""
echo "最近5个上游提交:"
git log develop..local-upstream/develop --oneline -5
echo ""

read -p "继续合并? (y/n) " -n 1 -r
echo
if [[ ! $REPLY =~ ^[Yy]$ ]]; then
    echo "合并已取消"
    exit 0
fi

# 5. 执行合并
echo -e "${YELLOW}[5/7] 执行合并...${NC}"
git merge local-upstream/develop --no-commit || true

# 检查冲突数量
CONFLICT_COUNT=$(git status | grep -c "both modified" || echo "0")
echo "冲突文件数: $CONFLICT_COUNT"

# 6. 自动处理冲突
echo -e "${YELLOW}[6/7] 自动处理冲突...${NC}"

# 6.1 采用上游版本的文件
echo "  - 采用上游版本..."
git checkout --theirs apps/frontend/src/main/terminal/ 2>/dev/null || true
git checkout --theirs apps/frontend/src/renderer/stores/task-store.ts 2>/dev/null || true
git checkout --theirs apps/frontend/src/renderer/stores/terminal-store.ts 2>/dev/null || true
git checkout --theirs apps/frontend/src/shared/state-machines/ 2>/dev/null || true
git checkout --theirs apps/frontend/src/main/pr-review-state-manager.ts 2>/dev/null || true

# 6.2 保留本地版本的文件
echo "  - 保留本地版本..."
git checkout --ours docs/中文/ 2>/dev/null || true
git checkout --ours docs/道劫/ 2>/dev/null || true
git checkout --ours docs/小说模块/ 2>/dev/null || true
git checkout --ours docs/cc-wf-studio/ 2>/dev/null || true
git checkout --ours docs/编辑器/ 2>/dev/null || true
git checkout --ours docs/侧边栏折叠/ 2>/dev/null || true
git checkout --ours docs/打包/ 2>/dev/null || true
git checkout --ours docs/代理终端/ 2>/dev/null || true
git checkout --ours docs/任务/ 2>/dev/null || true
git checkout --ours docs/Coding/ 2>/dev/null || true
git checkout --ours docs/Planning/ 2>/dev/null || true
git checkout --ours docs/Validation/ 2>/dev/null || true
git checkout --ours docs/git/ 2>/dev/null || true
git checkout --ours docs/hook/ 2>/dev/null || true
git checkout --ours docs/App名字/ 2>/dev/null || true
git checkout --ours CLA.zh-CN.md 2>/dev/null || true
git checkout --ours pnpm-lock.yaml 2>/dev/null || true

# 6.3 添加已处理的文件
git add apps/frontend/src/main/terminal/
git add apps/frontend/src/renderer/stores/
git add apps/frontend/src/shared/state-machines/
git add docs/
git add CLA.zh-CN.md pnpm-lock.yaml

# 7. 检查剩余冲突
echo -e "${YELLOW}[7/7] 检查剩余冲突...${NC}"
REMAINING=$(git status | grep "both modified" || true)
if [ -n "$REMAINING" ]; then
    echo -e "${RED}以下文件仍需手动处理:${NC}"
    echo "$REMAINING"
    echo ""
    echo "请参考 docs/合并上游代码/高冲突文件处理.md 处理这些文件"
else
    echo -e "${GREEN}✓ 所有冲突已自动处理${NC}"
fi

echo ""
echo "========================================"
echo "合并准备完成!"
echo "========================================"
echo "备份分支: $BACKUP_BRANCH"
echo ""
echo "下一步:"
echo "1. 手动处理剩余冲突文件 (如果有)"
echo "2. 运行: git status 确认"
echo "3. 运行: npm run build 验证"
echo "4. 运行: git commit 提交"
```

## 2. 验证脚本

保存为 `verify-merge.sh`：

```bash
#!/bin/bash
# 验证合并结果

echo "========================================"
echo "合并验证脚本"
echo "========================================"

cd /Users/zhengbingjin/Project/Github/Auto-Claude

# 1. 检查冲突标记
echo "[1/5] 检查冲突标记..."
CONFLICTS=$(grep -r "<<<<<<" apps/ --include="*.ts" --include="*.tsx" --include="*.json" 2>/dev/null || true)
if [ -n "$CONFLICTS" ]; then
    echo "❌ 发现冲突标记:"
    echo "$CONFLICTS"
else
    echo "✓ 无冲突标记"
fi

# 2. 检查本地独有功能
echo "[2/5] 检查本地独有功能..."
if [ -d "apps/frontend/src/renderer/components/workflow" ]; then
    echo "✓ 工作流工作室存在"
else
    echo "❌ 工作流工作室缺失!"
fi

if [ -d "apps/frontend/src/renderer/components/novel" ]; then
    echo "✓ 小说模块存在"
else
    echo "❌ 小说模块缺失!"
fi

if [ -d "apps/backend/novel" ]; then
    echo "✓ 小说后端存在"
else
    echo "❌ 小说后端缺失!"
fi

# 3. 检查状态机
echo "[3/5] 检查 XState 状态机..."
STATE_MACHINES=(
    "apps/frontend/src/shared/state-machines/task-machine.ts"
    "apps/frontend/src/shared/state-machines/terminal-machine.ts"
    "apps/frontend/src/shared/state-machines/pr-review-machine.ts"
    "apps/frontend/src/shared/state-machines/roadmap-generation-machine.ts"
)

for machine in "${STATE_MACHINES[@]}"; do
    if [ -f "$machine" ]; then
        echo "✓ $(basename $machine) 存在"
    else
        echo "❌ $(basename $machine) 缺失!"
    fi
done

# 4. 检查本地文档
echo "[4/5] 检查本地文档..."
DOCS=("docs/中文" "docs/道劫" "docs/小说模块" "docs/cc-wf-studio")
for doc in "${DOCS[@]}"; do
    if [ -d "$doc" ]; then
        echo "✓ $doc 存在"
    else
        echo "❌ $doc 缺失!"
    fi
done

# 5. Git 状态
echo "[5/5] Git 状态..."
git status --short

echo ""
echo "========================================"
echo "验证完成"
echo "========================================"
```

## 3. 回滚脚本

保存为 `rollback-merge.sh`：

```bash
#!/bin/bash
# 回滚合并

echo "========================================"
echo "合并回滚脚本"
echo "========================================"

cd /Users/zhengbingjin/Project/Github/Auto-Claude

# 列出备份分支
echo "可用的备份分支:"
git branch | grep backup

echo ""
read -p "输入要回滚到的备份分支名: " BACKUP_BRANCH

if git show-ref --verify --quiet "refs/heads/$BACKUP_BRANCH"; then
    echo "回滚到 $BACKUP_BRANCH..."
    git reset --hard "$BACKUP_BRANCH"
    echo "✓ 回滚完成"
else
    echo "❌ 分支不存在: $BACKUP_BRANCH"
    exit 1
fi
```

## 4. 使用方法

```bash
# 1. 复制脚本到项目根目录
cd /Users/zhengbingjin/Project/Github/Auto-Claude

# 2. 添加执行权限
chmod +x merge-upstream.sh verify-merge.sh rollback-merge.sh

# 3. 执行合并
./merge-upstream.sh

# 4. 验证结果
./verify-merge.sh

# 5. 如需回滚
./rollback-merge.sh
```
